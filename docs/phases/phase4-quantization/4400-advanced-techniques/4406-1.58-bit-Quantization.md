## Abstract

The frontier of extreme quantization: representing weights with just 1.58 bits on average using {-1, 0, +1} encoding.

## What is 1.58-bit?

**Mathematical Meaning:**
- **Not literally 1.58 bits** (you can't have fractional bits)
- **Average entropy:** 1.58 bits per weight
- **3 values:** {-1, 0, +1} ternary representation
- **Distribution:** P(-1) = 0.25, P(0) = 0.5, P(+1) = 0.25

**Entropy Calculation:**
```text
H = -Σ p(x) × log₂(p(x))
H = -(0.25 × log₂(0.25) + 0.5 × log₂(0.5) + 0.25 × log₂(0.25))
H = -(0.25 × -2 + 0.5 × -1 + 0.25 × -2)
H = -(-0.5 - 0.5 - 0.5) = 1.58 bits
```

## Ternary Quantization Methods

### 1. Deterministic Ternarization

```python
import torch
import numpy as np

def ternarize_deterministic(weight, threshold=0.05):
    """
    Convert weights to {-1, 0, +1}.

    Args:
        weight: Original weight tensor
        threshold: Pruning threshold (0.05 = 5% of max magnitude)

    Returns:
        Ternarized tensor
    """
    # Compute threshold as fraction of max weight
    delta = threshold * weight.abs().max()

    # Apply ternarization
    ternary = torch.zeros_like(weight)
    ternary[weight > delta] = 1
    ternary[weight < -delta] = -1
    # Values in [-delta, delta] become 0

    return ternary
```

### 2. Learned Threshold Ternarization

```python
class LearnedTernary(nn.Module):
    """Ternarization with learned threshold."""

    def __init__(self, weight):
        super().__init__()
        self.weight = nn.Parameter(weight)
        # Learnable threshold
        self.threshold = nn.Parameter(torch.tensor(0.05))

    def forward(self, x):
        # Straight-through estimator
        ternary_weight = ternarize_with_ste(self.weight, self.threshold)
        return nn.functional.linear(x, ternary_weight)

    def ternarize_with_ste(self, weight, threshold):
        """Ternarize with straight-through estimator."""

        # Forward: compute ternary values
        delta = threshold * weight.abs().max()
        ternary = torch.zeros_like(weight)
        ternary[weight > delta] = 1
        ternary[weight < -delta] = -1

        # Backward: gradient flows through ternary operation
        return (ternary - weight).detach() + weight
```

### 3. Probabilistic Ternarization

```python
def ternarize_probabilistic(weight, temperature=1.0):
    """
    Sample from {-1, 0, +1} with learned probabilities.

    Uses Gumbel-Softmax for differentiable sampling.
    """
    # Normalize weights to [0, 1]
    normalized = torch.sigmoid(weight / temperature)

    # Gumbel-Softmax trick for sampling
    logits = torch.stack([
        torch.log(normalized + 1e-10),      # P(+1)
        torch.log(1 - normalized + 1e-10)    # P(0) and P(-1)
    ], dim=-1)

    # Sample with Gumbel noise
    gumbel = -torch.log(-torch.log(torch.rand_like(logits) + 1e-10) + 1e-10)
    samples = torch.softmax((logits + gumbel) / temperature, dim=-1)

    # Convert to {-1, 0, +1}
    positive = samples[..., 0]
    ternary = torch.where(
        positive > 0.5,
        torch.ones_like(weight),
        torch.where(
            torch.rand_like(weight) > 0.5,
            torch.zeros_like(weight),
            -torch.ones_like(weight)
        )
    )

    return ternary
```

## Training 1.58-bit Models

### 1. Ternary Weight Networks (TWN)

```python
class TernaryLinear(nn.Module):
    """Linear layer with ternary weights."""

    def __init__(self, in_features, out_features):
        super().__init__()
        # Full-precision weights for training
        self.weight = nn.Parameter(
            torch.randn(out_features, in_features)
        )
        self.threshold = nn.Parameter(torch.tensor(0.05))

    def forward(self, x):
        # Ternarize weights
        ternary_w = self._ternarize(self.weight, self.threshold)
        return nn.functional.linear(x, ternary_w)

    def _ternarize(self, weight, threshold):
        """Ternarize with straight-through estimator."""
        delta = threshold * weight.abs().max()
        ternary = torch.zeros_like(weight)
        ternary[weight > delta] = 1
        ternary[weight < -delta] = -1

        # STE: forward uses ternary, backward uses full precision
        return (ternary - weight).detach() + weight

    def extra_repr(self):
        return f'in_features={self.weight.shape[1]}, out_features={self.weight.shape[0]}'
```

### 2. Learned Ternary Quantization (LTQ)

```python
class LTQLayer(nn.Module):
    """Learned Ternary Quantization layer."""

    def __init__(self, in_features, out_features, sparsity=0.5):
        super().__init__()

        # Main weights
        self.weight = nn.Parameter(
            torch.randn(out_features, in_features) * 0.01
        )

        # Learnable scaling factors
        self.scale_pos = nn.Parameter(torch.ones(1))
        self.scale_neg = nn.Parameter(torch.ones(1))

        # Target sparsity
        self.sparsity = sparsity

    def forward(self, x):
        # Quantize weights
        ternary_w = self._ltq_quantize(self.weight)

        # Apply separate scales for +1 and -1
        scaled_w = torch.where(
            ternary_w > 0,
            ternary_w * self.scale_pos,
            ternary_w * self.scale_neg
        )

        return nn.functional.linear(x, scaled_w)

    def _ltq_quantize(self, weight):
        """Learned ternary quantization."""

        # Compute percentile-based threshold
        w_flat = weight.abs().view(-1)
        threshold = torch.quantile(
            w_flat,
            self.sparsity
        )

        # Ternarize
        ternary = torch.zeros_like(weight)
        ternary[weight > threshold] = 1
        ternary[weight < -threshold] = -1

        # Straight-through estimator
        return (ternary - weight).detach() + weight
```

### 3. Full Training Loop

```python
def train_ternary_model(model, train_loader, epochs, lr=1e-3):
    """Train model with ternary weights."""

    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    criterion = nn.CrossEntropyLoss()

    for epoch in range(epochs):
        total_loss = 0
        correct = 0
        total = 0

        for batch_idx, (data, target) in enumerate(train_loader):
            data, target = data.cuda(), target.cuda()

            optimizer.zero_grad()

            # Forward pass with ternary weights
            output = model(data)
            loss = criterion(output, target)

            # Backward pass (STE enables gradients)
            loss.backward()
            optimizer.step()

            # Metrics
            total_loss += loss.item()
            pred = output.argmax(dim=1)
            correct += (pred == target).sum().item()
            total += target.size(0)

        # Print statistics
        avg_loss = total_loss / len(train_loader)
        accuracy = 100. * correct / total

        print(f'Epoch {epoch}: Loss={avg_loss:.4f}, Acc={accuracy:.2f}%')

        # Print sparsity statistics
        print_sparsity(model)
```

## Scaling to Large Models

### 1. Layer-wise Ternarization

```python
def ternarize_model_layerwise(model, thresholds=None):
    """
    Ternarize model with layer-specific thresholds.

    Different layers may need different sparsity levels.
    """
    if thresholds is None:
        thresholds = {
            'attention': 0.03,
            'ffn': 0.05,
            'output': 0.02
        }

    for name, module in model.named_modules():
        if isinstance(module, nn.Linear):
            # Determine threshold based on layer type
            if 'attention' in name:
                threshold = thresholds['attention']
            elif 'ffn' in name or 'mlp' in name:
                threshold = thresholds['ffn']
            else:
                threshold = thresholds['output']

            # Apply ternarization
            with torch.no_grad():
                ternary_w = ternarize_deterministic(
                    module.weight.data,
                    threshold
                )
                module.weight.data = ternary_w

    return model
```

```python
def progressive_ternarization(model, train_loader, total_steps):
```

## Deployment Considerations

### 1. Efficient Ternary Matrix Multiplication

```python
def ternary_matmul(ternary_a, ternary_b):
    """
    Optimized matrix multiplication for ternary matrices.

    Exploits structure: {-1, 0, +1}
    """
    # Separate positive and negative
    pos_a = (ternary_a > 0).float()
    neg_a = (ternary_a < 0).float()

    pos_b = (ternary_b > 0).float()
    neg_b = (ternary_b < 0).float()

    # Compute using bit operations (faster)
    # pos * pos = positive contribution
    # neg * neg = positive contribution
    # pos * neg or neg * pos = negative contribution

    result = (
        torch.matmul(pos_a, pos_b) +
        torch.matmul(neg_a, neg_b) -
        torch.matmul(pos_a, neg_b) -
        torch.matmul(neg_a, pos_b)
    )

    return result
```

### 2. Bit-Packed Storage

```python
def pack_ternary(ternary_tensor):
    """
    Pack ternary values into bits (2 bits per weight).

    Encoding:
    00 = -1
    01 = 0
    10 = +1
    11 = (unused)
    """
    # Map {-1, 0, +1} to {0, 1, 2}
    mapped = (ternary_tensor + 1).long()  # {-1,0,1} -> {0,1,2}

    # Pack 4 ternary values into 8 bits
    packed = torch.zeros(
        (ternary_tensor.numel() + 3) // 4,
        dtype=torch.uint8
    )

    for i in range(0, ternary_tensor.numel(), 4):
        chunk = mapped.flatten()[i:i+4]
        packed_val = 0
        for j, val in enumerate(chunk):
            packed_val |= val.item() << (2 * j)
        packed[i // 4] = packed_val

    return packed

# Compression ratio: 16x (from FP32)
```

## Accuracy Recovery Techniques

### 1. Knowledge Distillation

```python
def distill_to_ternary(teacher, student, train_loader, temperature=3.0):
    """Distill from full-precision teacher to ternary student."""

    kl_div = nn.KLDivLoss(reduction='batchmean')
    optimizer = torch.optim.Adam(student.parameters())

    for batch in train_loader:
        # Teacher predictions (soft targets)
        with torch.no_grad():
            teacher_logits = teacher(batch)
            teacher_probs = torch.softmax(teacher_logits / temperature, dim=-1)

        # Student predictions
        student_logits = student(batch)
        student_log_probs = torch.log_softmax(student_logits / temperature, dim=-1)

        # Distillation loss
        loss = kl_div(student_log_probs, teacher_probs) * (temperature ** 2)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
```

### 2. Bias Correction

```python
def correct_bias(model, calibration_loader):
    """Adjust biases to compensate for quantization error."""

    # Collect activation statistics
    activations = []
    with torch.no_grad():
        for batch in calibration_loader:
            output = model(batch)
            activations.append(output)

    # Compute optimal bias correction
    mean_activation = torch.cat(activations).mean(dim=0)

    # Adjust output layer biases
    for module in model.modules():
        if isinstance(module, nn.Linear) and module.out_features == mean_activation.shape[0]:
            module.bias.data -= mean_activation * 0.1

    return model
```
